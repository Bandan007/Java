CSV Processing
import csv

# Step 1: Create the CSV file and write student records
with open("students.csv", "w", newline="") as file:
    writer = csv.writer(file)

    # Write column headings
    writer.writerow(["StudentID", "Name", "Marks"])

    # Write student records
    writer.writerow(["S001", "Rahul", 82])
    writer.writerow(["S002", "Priya", 68])
    writer.writerow(["S003", "Amit", 91])
    writer.writerow(["S004", "Sneha", 74])
    writer.writerow(["S005", "Rohan", 79])

# Step 2: Read the CSV file and process the records
total_marks = 0
student_count = 0

print("Students who scored more than 75 marks:")
print("-----------------------------------------")

with open("students.csv", "r", newline="") as file:
    reader = csv.DictReader(file)

    for student in reader:
        marks = int(student["Marks"])

        # Calculate total marks
        total_marks += marks
        student_count += 1

        # Display students scoring more than 75
        if marks > 75:
            print(student["Name"])

# Step 3: Calculate and display the average marks
average_marks = total_marks / student_count

print("-----------------------------------------")
print("Average Marks:", average_marks)
Copy File
# Take source and destination filenames as input
source_file = input("Enter the source filename: ")
destination_file = input("Enter the destination filename: ")

try:
    # Open the source file in read mode
    with open(source_file, "r") as source:
        content = source.read()

    # Open the destination file in write mode
    with open(destination_file, "w") as destination:
        destination.write(content)

    print("File copied successfully.")
    print("Source file:", source_file)
    print("Destination file:", destination_file)

except FileNotFoundError:
    print("Error: The source file was not found.")
    print("Please check the filename and try again.")

except PermissionError:
    print("Error: Permission denied while accessing the file.")

except Exception as e:
    print("An unexpected error occurred:", e)
Count Vowels
# Open the existing file in read mode
with open("paragraph.txt", "r") as file:
    content = file.read()

# Define vowels
vowels = "aeiouAEIOU"

vowel_count = 0
consonant_count = 0

# Examine each character in the file
for character in content:
    if character.isalpha():
        if character in vowels:
            vowel_count += 1
        else:
            consonant_count += 1

# Display the results
print("Total number of vowels:", vowel_count)
print("Total number of consonants:", consonant_count)

Krishnamurthy Number
num = int(input("Enter a number: "))
temp = num
sum = 0

while temp > 0:
    digit = temp % 10
    fact = 1
    for i in range(1, digit + 1):
        fact = fact * i
    sum = sum + fact
    temp = temp // 10

if num == sum:
    print("Krishnamurthy Number")
else:
    print("Not Krishnamurthy")

File Operations
# Create and write data into a file
f = open("sample.txt", "w")
f.write("Hello Python Students\n")
f.write("File handling is easy\n")
f.write("Practice makes perfect")
f.close()

# Open file in read mode
f = open("sample.txt", "r")

# f.tell() - shows current position of file pointer
print("Position after opening:", f.tell())

# f.read() - reads whole file
content = f.read()
print("Content of file:")
print(content)

print("Position after reading:", f.tell())

# Counting occurrences using count() on the read string
print("Count of 'e' in file:", content.count("e"))
print("Count of 'Python' in file:", content.count("Python"))

# f.seek() - moves file pointer to a position
f.seek(0)                    # go back to start
print("\nAfter seek(0), position is:", f.tell())
print("First 5 characters:", f.read(5))

f.seek(6)                    # move to 6th position
print("\nAfter seek(6), position is:", f.tell())
print("Next 6 characters:", f.read(6))

# f.readline() - reads one line at a time
f.seek(0)
print("\nUsing readline():")
print(f.readline(), end="")

# f.readlines() - reads all lines into a list
f.seek(0)
lines = f.readlines()
print("\nUsing readlines():")
print(lines)
print("Total number of lines:", len(lines))

f.close()
print("\nFile closed successfully")
