Fibonacci
import java.util.Scanner;

public class Fibonacci {
    // Recursive method to return nth Fibonacci number
    static int fibRecursive(int n) {
        if (n <= 1) return n;
        return fibRecursive(n - 1) + fibRecursive(n - 2);
    }

    // Non‑recursive (iterative) method
    static void fibNonRecursive(int n) {
        int a = 0, b = 1, c;
        System.out.print("Fibonacci series (non‑recursive): ");
        for (int i = 0; i < n; i++) {
            System.out.print(a + " ");
            c = a + b;
            a = b;
            b = c;
        }
        System.out.println();
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter number of terms: ");
        int n = sc.nextInt();
        
        // Recursive
        System.out.print("Fibonacci series (recursive): ");
        for (int i = 0; i < n; i++) {
            System.out.print(fibRecursive(i) + " ");
        }
        System.out.println();

        // Non‑recursive
        fibNonRecursive(n);
        sc.close();
    }
}
Inheritance & overriding
class Vehicle {
    void calculateMileage() {
        System.out.println("Generic vehicle mileage: not defined.");
    }
}

class Car extends Vehicle {
    @Override
    void calculateMileage() {
        System.out.println("Car mileage: 15 km/l");
    }
}

class Bike extends Vehicle {
    @Override
    void calculateMileage() {
        System.out.println("Bike mileage: 40 km/l");
    }
}

public class VehicleDemo {
    public static void main(String[] args) {
        Vehicle v1 = new Car();
        Vehicle v2 = new Bike();
        v1.calculateMileage();  // Car mileage
        v2.calculateMileage();  // Bike mileage
    }
}
Array Reverse
import java.util.Scanner;

public class ReverseArray {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter number of elements: ");
        int n = sc.nextInt();
        int[] arr = new int[n];
        System.out.println("Enter " + n + " numbers:");
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        System.out.println("Array in reverse order:");
        for (int i = n - 1; i >= 0; i--) {
            System.out.print(arr[i] + " ");
        }
        sc.close();
    }
}
Palindrome Check
        import java.util.Scanner;

public class PalindromeCheck {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter a string: ");
        String str = sc.nextLine();
        String rev = new StringBuilder(str).reverse().toString();
        if (str.equalsIgnoreCase(rev)) {
            System.out.println(str + " is a palindrome.");
        } else {
            System.out.println(str + " is not a palindrome.");
        }
        sc.close();
    }
}

Employee class
import java.util.Scanner;

class Employee {
    int id;
    String name, department, phone;
    double salary;

    Employee(int id, String name, String department, String phone, double salary) {
        this.id = id;
        this.name = name;
        this.department = department;
        this.phone = phone;
        this.salary = salary;
    }

    void display() {
        System.out.printf("%-5d %-20s %-15s %-12s %.2f\n", id, name, department, phone, salary);
    }
}

public class EmployeeDetails {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter number of employees: ");
        int n = sc.nextInt();
        Employee[] employees = new Employee[n];
        for (int i = 0; i < n; i++) {
            System.out.println("Enter details for employee " + (i + 1) + ":");
            System.out.print("ID: ");
            int id = sc.nextInt();
            sc.nextLine();  // consume newline
            System.out.print("Name: ");
            String name = sc.nextLine();
            System.out.print("Department: ");
            String dept = sc.nextLine();
            System.out.print("Phone: ");
            String phone = sc.nextLine();
            System.out.print("Salary: ");
            double sal = sc.nextDouble();
            employees[i] = new Employee(id, name, dept, phone, sal);
        }
        System.out.println("\nEmployee Details:");
        System.out.printf("%-5s %-20s %-15s %-12s %s\n", "ID", "NAME", "DEPARTMENT", "PHONE", "SALARY");
        for (Employee e : employees) {
            e.display();
        }
        sc.close();
    }
}

User defined exception UnderAge
import java.util.Scanner;

class UnderAge extends Exception {
    public UnderAge(String message) {
        super(message);
    }
}

public class AgeValidation {
    static void validateAge(int age) throws UnderAge {
        if (age < 18) {
            throw new UnderAge("Age " + age + " is under 18. Not allowed.");
        } else {
            System.out.println("Age " + age + " is valid.");
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter your age: ");
        int age = sc.nextInt();
        try {
            validateAge(age);
        } catch (UnderAge e) {
            System.out.println("Exception: " + e.getMessage());
        }
        sc.close();
    }
}
